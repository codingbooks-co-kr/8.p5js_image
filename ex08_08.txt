// 컬러 이미지의 픽셀값을 읽고 그레이스케일 이미지로 출력 (입·출력 이미지 모두 표시)

let img0, img1;

function preload() {
  img0 = loadImage('landscape.jpg');
}

function setup() {
  createCanvas(img0.width*2, img0.height);
  img1 = createImage(img0.width, img0.height);	// 출력 이미지를 위한 객체 생성
  noLoop();						// draw() 반복 실행 정지
}

function draw() {
  img0.loadPixels();			// 이미지(img0)의 pixels[]에 접근하기 전 픽셀값 로드
  img1.loadPixels();			// 이미지(img1)의 pixels[]에 접근하기 전 픽셀값 로드
  for (let y = 0; y < img0.height; y++) {
    for (let x = 0; x < img0.width; x++) {
      let index = (x + y * img0.width) * 4;
      let r = img0.pixels[index+0]; 		// Red값 읽기
      let g = img0.pixels[index+1];		// Green값 읽기
      let b = img0.pixels[index+2]; 		// Blue값 읽기

      let gray = (r + g + b) / 3;		// RGB 컬러값의 평균(→그레이스케일 밝기값)

      img1.pixels[index+0] = gray;		// Red값 쓰기
      img1.pixels[index+1] = gray;		// Green값 쓰기
      img1.pixels[index+2] = gray;		// Blue값 쓰기
      img1.pixels[index+3] = 255;		// Alpha (투명: 0, 불투명: 255)
    }
  }
  img1.updatePixels();				// 이미지(img1)의 변경된 pixels[] 업데이트
  image(img0, 0, 0);				// 원본 이미지 표시
  image(img1, img0.width, 0);			// 출력 이미지 표시
}
