// 이미지의 특정 픽셀 영역 읽기

let img;

function preload() {
  img = loadImage('landscape.jpg');
}

function setup() {
  createCanvas(img.width*2, img.height);
}

function draw() {
  background(0);
  image(img, 0, 0);			  // 원본 이미지 표시
  
  let x = 50, y = 0, w = 100, h = 100;
  let imgCopy = get(x, y, w, h);	  // 읽은 픽셀 영역 반환
  image(imgCopy, img.width, 0);	  // 읽은 픽셀 영역을 캔버스 상에 표시

  noFill();
  stroke(255, 0, 0);
  rect(x, y, w, h);			  // 읽은 픽셀 영역을 사각형으로 표시
}
