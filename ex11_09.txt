// blendMode()를 사용한 비디오 프레임과 원 도형의 블랜딩

let video;

function setup() {
  createCanvas(400, 300);
  video = createCapture(VIDEO);		// 실시간 비디오 요소 생성
  video.hide();					// 비디오 요소 숨김
}

function draw() {
  clear();					// 캔버스 비우기
  if (mouseIsPressed) blendMode(DIFFERENCE);
  else blendMode(BLEND);
  image(video, 0, 0, width, height);		// 캔버스에 비디오 프레임 표시
  
  noStroke();
  fill(0, 255, 0);
  ellipse(mouseX, mouseY, 200, 200);		// 마우스를 따라오는 원형
}
