// NORMAL 모드

let img;

function preload() {
  img = loadImage('landscape.jpg');
}

function setup() {
  createCanvas(400, 400, WEBGL);
}

function draw() {
  background(0);
  textureMode(NORMAL);  // 텍스처 모드
  texture(img);		    // 텍스처 설정
  beginShape();
  vertex(-200, -200, 0, 0);
  vertex(200, -200, 1, 0);
  vertex(100, 200, 1, 1);
  vertex(-100, 200, 0, 1);
  endShape();
}
