// 점과 선 그리기

function setup() {
  createCanvas(300, 300);	// 캔버스 크기를 가로*세로=300*300 픽셀로 설정
  background(200);		// 배경색을 200으로 설정

  strokeWeight(5);		// 점과 선의 굵기 설정: 5픽셀
  point(150, 120);		// 좌표 (150, 120)에 점 표시
  point(150, 180);		// 좌표 (150, 180)에 점 표시
  line(0, 0, 300, 300);	// 좌표 (0, 0)에서 (300, 300)까지 선 그리기
  line(300, 0, 0, 300);	// 좌표 (300, 0)에서 (0, 300)까지 선 그리기
}
