// 원의 내부를 다양한 색상으로 표현

function setup() {
  createCanvas(600, 200);
  background(200);		// 0(검정)∼255(흰색) 범위로 배경색 설정

  noFill();                 	// 도형의 내부 색상 없음
  ellipse(100, 100, 120, 120); 

  fill(0);              	// 0(검정)∼255(흰색)의 범위로 도형의 내부 색상 설정
  ellipse(200, 100, 120, 120);

  fill(255, 0, 0);          	// fill(r, g, b)에서 빨강(red) 값을 최대치 255로 설정
  ellipse(300, 100, 120, 120);

  fill(0, 255, 0);          	// fill(r, g, b)에서 녹색(green) 값을 최대치 255로 설정
  ellipse(400, 100, 120, 120);

  fill(0, 0, 255);          	// fill(r, g, b)에서 파랑(blue) 값을 최대치 255로 설정
  ellipse(500, 100, 120, 120);
}
