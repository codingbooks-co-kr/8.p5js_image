// 이미지 좌표 이동 및 좌표 회전(라디안)

let img, theta = 0;

function preload() {
  img = loadImage('landscape.jpg');
}

function setup() {
  createCanvas(400, 400);
  angleMode(RADIANS);	// 라디안
  imageMode(CENTER);
}

function draw() { 
  background(0);
  translate(width/2, height/2);	// 좌표이동
  rotate(theta);			// 좌표회전
  image(img, 0, 0, 200, 200);
}

function mousePressed() {
  theta += 0.1;		// 0.1라디안씩 누적
}
