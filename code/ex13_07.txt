// p5.js: 얼굴, 주먹, 손바닥 모델 기반 인식: 이미지 회전 제어
// modelURL을 사용자 스스로 만든 모델의 링크로 대체할 것!

let modelURL = 'https://teachablemachine.withgoogle.com/models/k87u0EAN-/';
let img, classifier, video, flippedVideo, d, label = '모델 로딩 중', degree = 0;

function preload() {
  img = loadImage('landscape.jpg');
  classifier = ml5.imageClassifier(modelURL + 'model.json');
}

function setup() {
  createCanvas(400, 300, WEBGL);	// 3차원 렌더러 사용
  video = createCapture(VIDEO);	// 실시간 비디오 요소 생성
  video.size(200, 150);			// 비디오 요소의 크기 설정
  classifyVideo();			// getResult() 호출 → 인식 결과 출력

  d = createDiv('');
  d.style('font-size', '20px');		// 폰트 크기(20픽셀) 설정

  angleMode(DEGREES);		// DEGREES 모드 설정 (0o∼360o 범위)
  imageMode(CENTER);		// 이미지 정렬 모드: 프레임의 중심에서 폭, 높이 설정
}

function draw() {
  background(0);
  rotateX(degree);			// x축 방향 회전
  image(img, 0, 0);			// 캔버스에 이미지 표시
  
  d.html(label);				// 인식 결과 표시
  if (label === '얼굴') {			// 얼굴 인식 시
    degree += 0;			// 이미지 x축 회전 정지
  } else if (label === '주먹') {		// 주먹 인식 시
    degree += 1;			// 이미지 x축 정방향 회전
  } else if (label === '손바닥') {		// 손바닥 인식 시
    degree -= 1;			// 이미지 x축 역방향 회전
  }
}

function classifyVideo() {
  flippedVideo = ml5.flipImage(video);		// 수평으로 반전된 이미지
  classifier.classify(flippedVideo, getResult);	// flippedVideo가 주어지면 getResult() 호출
  flippedVideo.remove();			// 생성된 flippedVideo 요소 제거
}

function getResult(error, results) {		// 콜백함수
  if (error) {					// 인식 오류 시
    console.log(error);
    return;					// 함수 실행 종료
  } else {					// 인식 성공 시
    label = results[0].label;			// 인식 결과
    classifyVideo();				// getResult() 재호출함으로써 반복 인식
  }
}
