// 이미지의 좌우 측면을 마우스로 누르면 이미지가 파노라마 형식으로 이동하면서 출력

let img, x;

function preload() {
  img = loadImage('mountain_lake.jpg');
}

function setup() {
  createCanvas(300, 300);
  x = img.width / 2;				// 실행 초기, 이미지의 가운데 부분 출력 
}

function draw() {
  background(0);
  image(img, width/2-x, 0);
  if (mouseIsPressed) {			// 마우스를 누른 경우
    if (mouseX < 0.2 * width) {		// 캔버스 너비의 왼쪽 20%를 누른 경우
      x -= 5;
    } else if (mouseX > 0.8 * width) {		// 캔버스 너비의 오른쪽 20%를 누른 경우
      x += 5;
    }
  }
}
