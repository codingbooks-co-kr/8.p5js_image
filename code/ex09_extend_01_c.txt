// HSB 컬러모드: 슬라이더로 HSB 컬러 변경

let d, slider1, slider2, slider3;

function setup() {
  createCanvas(300, 300);
  colorMode(HSB, 255);		// HSB 컬러모드

  slider1 = createSlider(0, 255, 0);	// 슬라이더 생성 (H 범위:0∼255, 초깃값:0)
  slider1.position(0, height+30);	// 슬라이더 위치 설정
  
  slider2 = createSlider(0, 255, 0);	// 슬라이더 생성 (S 범위:0∼255, 초깃값:0)
  slider2.position(0, height+50);	// 슬라이더 위치 설정
  
  slider3 = createSlider(0, 255, 0);	// 슬라이더 생성 (B 범위:0∼255, 초깃값:0)
  slider3.position(0, height+70);	// 슬라이더 위치 설정
  
  d = createDiv('');
  d.style('font-size', '20px');		// 텍스트 폰트 크기를 (20픽셀로) 설정
}

function draw() {
  let h = slider1.value();		// 슬라이더값(h) 읽기
  let s = slider2.value();		// 슬라이더값(s) 읽기
  let b = slider3.value();		// 슬라이더값(b) 읽기
  background(h, s, b);			// 배경색에 HSB 슬라이더값 적용
  d.html('H:' + nf(h, 3) + ',  S:' + nf(s, 3) + ',  B:' + nf(b, 3));
}
