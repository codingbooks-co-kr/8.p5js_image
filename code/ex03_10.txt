// 키보드 상의 숫자를 눌러 원의 색상 변경

function setup() {
  createCanvas(300, 300);
}

function draw() {
  background(0);
  if (keyIsPressed && (key === '1')) {
    fill(255, 0, 0);
  } else if (keyIsPressed && (key === '2')) {
    fill(0, 255, 0);
  } else {
    fill(0, 0, 255);
  }
  ellipse(width/2, height/2, 200, 200);
}
