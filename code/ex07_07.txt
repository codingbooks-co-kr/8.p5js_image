// 이미지의 RGB 픽셀값을 읽고 색상 요소를 원의 색상으로 표현

let img, d;

function preload() {
  img = loadImage('landscape.jpg');
}

function setup() {
  createCanvas(img.width*2, img.height);
  d = createDiv('');
  d.style('font-size', '20px');		// 텍스트 폰트 크기를 (20픽셀로) 설정
}

function draw() {
  image(img, 0, 0);			// 원본 이미지 표시
  let c = get(mouseX, mouseY);	// 마우스 현재 위치 좌표의 색상값 읽기
  let r = red(c);			// 색상값 중 red 요소 읽기
  let g = green(c);			// 색상값 중 green 요소 읽기
  let b = blue(c);			// 색상값 중 blue 요소 읽기
  d.html('R:' + nf(r, 3) + ',  G:' + nf(g, 3) + ',  B:' + nf(b, 3)); 
  fill(r, 0, 0);         			// red 색상 표시
  ellipse(img.width+50, height/2, 50, 50);  
  fill(0, g, 0);         			// green 색상 표시
  ellipse(img.width+100, height/2, 50, 50);  
  fill(0, 0, b);         			// blue 색상 표시
  ellipse(img.width+150, height/2, 50, 50);  
}
