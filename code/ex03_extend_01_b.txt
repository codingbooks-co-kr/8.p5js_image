function setup() {
  print(1 == 1);	// 값이 같으므로 true
  print(1 === 1);	// 값과 데이터형이 같으므로 true

  print(1 == '1');	// 문자열은 숫자로 형변환된 후 비교
  print(1 == true);	// true는 숫자 1로 형변환된 후 비교
  print(0 == '0');	// 숫자 문자열은 숫자로 형변환된 후 비교
  print(0 == false);	// false는 숫자 0으로 형변환된 후 비교
  print('' == false);	// 빈 문자열('')은 false로 형변환된 후 비교
  print([] == false);	// 빈 배열([])은 false로 형변환된 후 비교

  print(1 === '1');	// 값과 데이터형이 다르므로 false
  print(1 === true);	// 이하 동일
  print(0 === '0');
  print(0 === false);
  print('' === false);
  print([] === false);
}
