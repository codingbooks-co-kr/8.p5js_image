// 슬라이더를 드래그함으로써 비디오 영상과 이미지의 상호 변환

let img, video, slider;

function preload() {
  img = loadImage('landscape.jpg');
}

function setup() {
  createCanvas(400, 300);
  video = createCapture(VIDEO);		// 실시간 비디오 요소 생성
  video.hide();					// 비디오 요소 숨김
  slider = createSlider(0, 255, 0, 0.1);		// 범위(0∼255), 초깃값(0), 스텝크기(0.1)
  slider.size(width);				// 슬라이더 크기 설정
}

function draw() {
  let val = slider.value();		// 슬라이더값 읽기

  tint(255, 255-val);			// 슬라이더를 오른쪽으로 드래그: 투명도는 0에 근접
  image(video, 0, 0, width, height);	// 비디오 프레임(video)이 투명하게 사라짐

  tint(255, val);				// 슬라이더를 오른쪽으로 드래그: val은 255에 근접
  image(img, 0, 0, width, height);	// 이미지(img)가 불투명하게 드러남
}
