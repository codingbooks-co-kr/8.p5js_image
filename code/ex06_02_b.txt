// 이미지 좌표 이동 및 좌표 회전(각도)

let img, degree = 0;

function preload() {
  img = loadImage('landscape.jpg');
}

function setup() {
  createCanvas(400, 400);
  angleMode(DEGREES);	// 각도
  imageMode(CENTER);
}

function draw() { 
  background(0);
  translate(width/2, height/2);	// 좌표이동
  rotate(degree);		// 좌표회전
  image(img, 0, 0, 200, 200);
}

function mousePressed() {
  degree += 6;		// 6도씩 누적
}
