// 마우스의 좌우 움직임에 따라 동영상 프레임과 이미지의 상호 변환

let img, video;

function preload() {
  img = loadImage('landscape.jpg');
}

function setup() {
  createCanvas(400, 300);
  video = createVideo('bird.mp4');	// 동영상 파일의 비디오 요소 생성
  video.hide();				// 비디오 요소 숨김
  video.loop();				// 동영상 반복 실행
}

function draw() {
  let x = constrain(mouseX, 0, width);
  x = map(x, 0, width, 0, 255);
  
  tint(255, 255-x);			// 마우스를 오른쪽으로 이동하면 투명도는 0에 근접
  image(video, 0, 0, width, height);	// 비디오 프레임(video)이 투명하게 사라짐

  tint(255, x);				// 마우스를 오른쪽으로 이동하면 x는 255에 근접
  image(img, 0, 0, width, height);	// 이미지(img)가 불투명하게 드러남
}
