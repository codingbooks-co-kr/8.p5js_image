// 숫자, 문자 및 문자열을 캔버스에 표시

function setup() {
  createCanvas(300, 300);		// 캔버스 크기를 가로*세로=300*300 픽셀로 설정
  background(200);			// 배경색을 200으로 설정

  textSize(30);            		// 문자 크기: 30픽셀
  text(7, 10, 50);        		// 숫자를 좌표 (10, 50)에 표시
  text('H', 10, 100);			// 문자를 좌표 (10, 100)에 표시
  text('Hello, p5.js!', 10, 150);	// 문자열을 좌표 (10, 150)에 표시
}
