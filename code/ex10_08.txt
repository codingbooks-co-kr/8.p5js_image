// 동영상 파일을 원형으로 재생

let video, val;

function setup() {
  createCanvas(400, 300);
  pixelDensity(1);			// 픽셀 밀도를 1로 설정
  video = createVideo('bird.mp4');	// 동영상 파일의 비디오 요소 생성
  video.size(width, height);		// 비디오 요소의 크기 설정
  video.hide();				// 비디오 요소 숨김
  video.loop();				// 동영상 반복 실행
}

function draw() {
  loadPixels();				// 캔버스의 pixels[]에 접근하기 전 픽셀값 로드
  video.loadPixels();			// 비디오(video)의 pixels[]에 접근하기 전 픽셀값 로드
  for (let y = 0; y < video.height; y++) {	// 중첩 for문 사용
    for (let x = 0; x < video.width; x++) {
      let index = (x + y * video.width) * 4;	// x행 y열의 픽셀 위치
      let r = video.pixels[index+0]; 		// Red값 읽기
      let g = video.pixels[index+1]; 		// Green값 읽기
      let b = video.pixels[index+2]; 		// Blue값 읽기

      let d = dist(x, y, width/2, height/2);	// (x,y) 좌표와 캔버스 중심간의 거리 측정
      if (d < height/2) val = 1;		// 거리(d)가 반지름(height/2) 미만: val=1
      else val = 0;				// 거리(d)가 반지름(height/2) 이상: val=0
      r *= val;					// r값은 0 또는 1
      g *= val;					// g값은 0 또는 1
      b *= val;					// b값은 0 또는 1

      pixels[index+0] = r;			// Red값 쓰기
      pixels[index+1] = g;			// Green값 쓰기
      pixels[index+2] = b;			// Blue값 쓰기
      pixels[index+3] = 255; 			// Alpha값 쓰기 (투명: 0, 불투명: 255)
    }
  }
  updatePixels();				// 캔버스의 변경된 pixels[] 업데이트
}
