// 슬라이더로 이미지의 HSB(→Hue, Saturation, Brightness) 색조 변경

let img, d, slider1, slider2, slider3;

function preload() {
  img = loadImage('landscape0.jpg');
}

function setup() {
  createCanvas(img.width, img.height);
  colorMode(HSB, 255);		// HSB 모드 (각 요소 범위: 0∼255)
  
  slider1 = createSlider(0, 255, 255);	// 슬라이더 생성 (H 범위:0∼255, 초깃값:255)
  slider1.position(0, img.height+30);	// 슬라이더 위치 설정
  slider1.size(img.width);		// 슬라이더 크기 설정
  
  slider2 = createSlider(0, 255, 255);	// 슬라이더 생성 (S 범위:0∼255, 초깃값:255)
  slider2.position(0, img.height+50);	// 슬라이더 위치 설정
  slider2.size(img.width);		// 슬라이더 크기 설정
  
  slider3 = createSlider(0, 255, 255);	// 슬라이더 생성 (B 범위:0∼255, 초깃값:255)
  slider3.position(0, img.height+70);	// 슬라이더 위치 설정
  slider3.size(img.width);		// 슬라이더 크기 설정
  
  d = createDiv('');
  d.style('font-size', '20px');		// 텍스트 폰트 크기를 (20픽셀로) 설정
}

function draw() {
  background(255);
  let h = slider1.value();		// 슬라이더값(h) 읽기
  let s = slider2.value();		// 슬라이더값(s) 읽기
  let b = slider3.value();		// 슬라이더값(b) 읽기

  tint(h, s, b);				// 이미지 색조에 HSB 슬라이더값 적용
  image(img, 0, 0);			// 이미지 표시

  d.html('H:' + nf(h, 3) + ',  S:' + nf(s, 3) + ',  B:' + nf(b, 3));
}
