// 동영상 프레임의 픽셀값을 랜덤하게 읽고 작은 점으로 재표현

let video, val;

function setup() {
  createCanvas(400, 300);
  video = createVideo('bird.mp4'); 	// 동영상 파일의 비디오 요소 생성
  video.size(width, height); 		// 비디오 요소의 크기 설정
  video.hide(); 				// 비디오 요소 숨김
  video.loop();				// 동영상 반복 실행
  background(0);
}

function draw() {
  video.loadPixels();			// 비디오(video)의 pixels[]에 접근하기 전 픽셀값 로드
  for (let i = 0; i < 100; i++) {		// 반복문으로 빠르게 처리
    let x = int(random(video.width));
    let y = int(random(video.height));
    let index = (x + y * video.width) * 4; 	// x행 y열의 픽셀 위치
    let r = video.pixels[index+0]; 		// Red값 읽기
    let g = video.pixels[index+1]; 		// Green값 읽기
    let b = video.pixels[index+2]; 		// Blue값 읽기

    stroke(r, g, b); 				// 픽셀값을 점의 색상으로 표현
    strokeWeight(5);				// 점 굵기(→5픽셀) 설정
    point(x, y); 				// 점으로 표시
  }
}
