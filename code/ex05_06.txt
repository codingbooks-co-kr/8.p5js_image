// 이미지 원운동

let img, theta = 0;

function preload() {
  img = loadImage('flower.jpg');
}

function setup() {
  createCanvas(500, 500);
  imageMode(CENTER);		// 이미지 정렬 모드: 이미지의 중심에서 폭,높이 설정
  background(0);
}

function draw() {
  let x = width/2 + 100 * cos(theta);		// 이미지의 x축 코사인 운동
  let y = height/2 + 100 * sin(theta);		// 이미지의 y축 사인 운동
  image(img, x, y);				// 이미지 표시
  theta += 0.01;
}
