// imageClassifier: 캔버스에 이미지 인식 결과 출력

let img, classifier, d;

function preload() {
  img = loadImage('puppy.jpg');
}

function setup() {
  createCanvas(img.width, img.height);
  classifier = ml5.imageClassifier('MobileNet', getModel);
  image(img, 0, 0);				// 캔버스에 이미지 표시
  d = createDiv('');
  d.style('font-size', '20px');			// 텍스트 폰트 크기(20픽셀) 설정
}

function getModel() {				// 콜백함수
  console.log('MobileNet 모델 로딩 완료 및 인식 시작!');
  classifier.classify(img, getResult);		// 이미지(img)가 주어지면 getResult() 호출
}

function getResult(error, results) {		// 콜백함수
  if (error) {					// 인식 오류 시
    console.log(error);
    return;					// 함수 실행 종료
  } else {					// 인식 성공 시
    console.log(results);				// 인식결과 출력(후보 3개)
    let label = results[0].label;				// 인식결과(제1후보)
    let confidence = int(results[0].confidence * 100);	// 신뢰도(제1후보)
    d.html(label + ' : ' + confidence + '%');		// 인식결과 및 신뢰도 출력
  }
}
