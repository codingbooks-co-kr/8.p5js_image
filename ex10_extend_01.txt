// 동영상 파일 재생/정지/속도/볼륨/재생위치 제어

let video, d, slider1, slider2, button, playing = false;

function setup() {
  createCanvas(400, 300);
  video = createVideo('train.mp4');	// 동영상 파일의 비디오 요소 생성
  video.hide();				// 비디오 요소 숨김
  video.loop();				// 동영상 반복 실행
  d = createDiv('');
  d.style('font-size', '20px');		// 텍스트 폰트 크기를 (20픽셀로) 설정

  slider1 = createSlider(0.1, 3.0, 1.0, 0.1);	// 범위(0.1∼3.0), 초깃값(1.0), 스텝크기(0.1)
  slider1.position(0, height+30); 	// 슬라이더 위치 설정
  slider1.size(width); 			// 슬라이더 크기 설정

  slider2 = createSlider(0.0, 1.0, 0.5, 0.01);	// 범위(0.0∼1.0), 초깃값(0.5), 스텝크기(0.01)
  slider2.position(0, height+50); 	// 슬라이더 위치 설정
  slider2.size(width); 			// 슬라이더 크기 설정

  button = createButton('정지'); 	// 버튼 생성 (버튼 레이블은 '정지')
  button.position(0, height+70); 	// 버튼 위치 설정
  button.size(100, 50); 		// 버튼 크기 설정
  button.style('font-size', '20px'); 	// 버튼 폰트 크기(20픽셀) 설정
  button.mousePressed(controlVideo); 	// 버튼 누르면 controlVideo() 자동 호출
}

function draw() {
  image(video, 0, 0, width, height);	// 캔버스에 동영상 프레임 표시
  let position = map(video.time(), 0, video.duration(), 0, width);
  fill(255, 0, 0);
  rect(0, 0, position, 15);		// 동영상 재생 위치를 사각형으로 표시

  let t1 = nf(video.time(), 0, 2);	// 동영상의 현재 재생 시간
  let t2 = nf(video.duration(), 0, 2);	// 동영상의 지속 시간

  let sp = slider1.value();
  video.speed(sp);			// 재생 속도 설정
  
  let vol = slider2.value();
  video.volume(vol);			// 재생 볼륨 설정
  d.html(t1 + '/' + t2 + '초' + ', 속도 = ' + sp + ', 볼륨 = ' + vol);
}

function mousePressed() {		// 마우스를 누를 때
  if (mouseX > 0 && mouseX < width && mouseY > 0 && mouseY < height) {
    let t = (mouseX / width) * video.duration();
    video.time(t);			// 동영상의 재생 시작 시간 이동
  }
}

function controlVideo() {		// 콜백함수: controlVideo()
  if (playing) {
    video.loop(); 			// 동영상 반복 실행
    button.html('정지');			// 버튼 레이블을 '정지'로 변경
  } else {
    video.pause(); 			// 동영상 정지
    button.html('재생');			// 버튼 레이블을 '재생'으로 변경
  }
  playing = !playing; 			// 토글
}
