// 두 개 이미지 좌표 이동 및 회전 (미완성)

let img, degree = 0;

function preload() {
  img = loadImage('landscape.jpg');
}

function setup() {
  createCanvas(400, 400);
  angleMode(DEGREES);
  imageMode(CENTER);
}

function draw() { 
  background(0);
  
  translate(width/4, height/4);	// 이동
  rotate(degree);		// 회전
  scale(0.5);			// 축소
  image(img, 0, 0, 200, 200);
  
  translate(width/2, height/2);	// 이동
  rotate(degree);		// 회전
  scale(0.5);			// 축소
  image(img, 0, 0, 200, 200);
}

function mousePressed() {
  degree += 6;
}
