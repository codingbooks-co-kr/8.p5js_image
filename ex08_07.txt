// 컬러 이미지에서 픽셀값을 읽고 그레이스케일 이미지로 출력 (출력 이미지만 표시)

let img;

function preload() {
  img = loadImage('landscape.jpg');
}

function setup() {
  createCanvas(img.width, img.height);
  noLoop();					// draw() 반복 실행 정지
}

function draw() {
  img.loadPixels();			// 이미지(img)의 pixels[]에 접근하기 전 픽셀값 로드
  for (let y = 0; y < img.height; y++) {	// 중첩 for문 사용
    for (let x = 0; x < img.width; x++) {
      let index = (x + y * img.width) * 4;	// x행 y열의 픽셀 위치
      let r = img.pixels[index+0]; 		// Red값 읽기
      let g = img.pixels[index+1]; 		// Green값 읽기
      let b = img.pixels[index+2]; 		// Blue값 읽기

      let gray = (r + g + b) / 3;		// RGB 컬러값의 평균(→그레이스케일 밝기값)

      img.pixels[index+0] = gray; 		// Red값 쓰기
      img.pixels[index+1] = gray; 		// Green값 쓰기
      img.pixels[index+2] = gray; 		// Blue값 쓰기
      img.pixels[index+3] = 255;		// Alpha값 쓰기 (투명: 0, 불투명: 255)
    }
  }
  img.updatePixels();				// 이미지(img)의 변경된 pixels[] 업데이트
  image(img, 0, 0);				// 변경된 이미지(img)를 캔버스에 표시
}
