// 마우스를 따라오는 이미지

let img;					// (이미지 객체) 변수 선언

function preload() {
  img = loadImage('flower.jpg');	// 이미지 파일 읽고 객체 생성
}

function setup() {
  createCanvas(500, 500);
  imageMode(CENTER);		// 이미지 정렬 모드: 이미지의 중심에서 폭,높이 설정
}

function draw() {
  background(0);
  image(img, mouseX, mouseY);	// 이미지를 마우스의 현재 위치 좌표로 이동
}
