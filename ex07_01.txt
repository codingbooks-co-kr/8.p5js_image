// 이미지의 특정 영역 복사

let img;

function preload() {
  img = loadImage('landscape.jpg');
}

function setup() {
  createCanvas(img.width*2, img.height);	// 캔버스 가로 크기는 이미지 크기의 두배
  console.log(img.width, img.height);		// 콘솔창에 이미지의 실제 크기를 출력
}

function draw() {
  background(0);
  image(img, 0, 0);				// 원본 이미지 표시

  let x = 50, y = 0, w = 100, h = 100;
  copy(x, y, w, h, img.width, 0, w, h);	// 이미지의 특정 영역 복사-붙여넣기

  noFill();
  stroke(255, 0, 0);
  rect(x, y, w, h);				// 원본 이미지의 복사 영역 표시
}
