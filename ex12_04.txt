// imageClassifier: 웹캠 영상으로 사물 인식

let classifier, video, d;

function setup() {
  createCanvas(640, 480);
  video = createCapture(VIDEO);	// 실시간 비디오 요소 생성
  video.hide();				// 비디오 요소 숨김
  classifier = ml5.imageClassifier('MobileNet', video, getModel);   // ❶getModel() 호출

  d = createDiv('MobileNet 모델 로딩 중!');
  d.style('font-size', '30px');
}

function draw() {
  image(video, 0, 0, width, height);	// 캔버스에 비디오 프레임 표시
}

function getModel() {			// 콜백함수
  console.log('MobileNet 모델 로딩 완료 및 인식 시작!');
  classifier.classify(getResult);		// ❷getResult() 호출
}

function getResult(error, results) {	// 콜백함수
  if (error) {				// 인식 오류 시
    console.log(error);
    return;				// 함수 실행 종료
  } else {				// 인식 성공 시
    let label = results[0].label;					// ❸인식결과
    let confidence = int(results[0].confidence * 100);		// 신뢰도
    d.html(label + ' : ' + confidence + '%');
    classifier.classify(getResult);			// ❹getResult() 호출
  }
}
