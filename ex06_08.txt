// 10개 이미지 각각 랜덤 좌표 이동 후 제자리 회전 및 축소

let img, num = 10, degree = 0, x = [], y = [];

function preload() {
  img = loadImage('landscape.jpg');
}

function setup() {
  createCanvas(500, 500);
  angleMode(DEGREES);		// DEGREES 모드 설정 (0o∼360o 범위)
  imageMode(CENTER);		// 이미지 정렬 모드: 이미지의 중심에서 폭, 높이 설정
  for (let i = 0; i < num; i++) {
    x[i] = random(width);		// 이동할 x좌표값 랜덤 설정
    y[i] = random(height);		// 이동할 y좌표값 랜덤 설정
  }
}

function draw() {
  background(0);
  for (let i = 0; i < num; i++) {
    push();
    translate(x[i], y[i]);			// 좌표 이동
    rotate(degree);			// 좌표 회전
    scale(0.9);				// 좌표 축소
    image(img, 0, 0, 100, 100);
    pop();
  }
  degree++;				// 1도씩 증가
}
